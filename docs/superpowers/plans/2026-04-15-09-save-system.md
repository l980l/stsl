## Task 1: SaveManager Autoload

- [ ] **Step 1: save_manager.gd 생성**

```gdscript
# autoload/save_manager.gd
class_name SaveManagerClass
extends Node

const SAVE_PATH := "user://save.json"

func save() -> void:
    var data := {
        "version": 1,
        "game_manager": GameManager.to_dict(),
        "team_manager": TeamManager.to_dict(),
        "deck_manager": DeckManager.to_dict(),
        "relics": _serialize_relics(),
    }
    var file := FileAccess.open(SAVE_PATH, FileAccess.WRITE)
    if file:
        file.store_string(JSON.stringify(data))
        file.close()

func load_save() -> bool:
    if not FileAccess.file_exists(SAVE_PATH):
        return false
    var file := FileAccess.open(SAVE_PATH, FileAccess.READ)
    if not file:
        return false
    var text := file.get_as_text()
    file.close()
    var data = JSON.parse_string(text)
    if data == null or not data.has("version"):
        return false
    GameManager.from_dict(data["game_manager"])
    TeamManager.from_dict(data["team_manager"])
    DeckManager.from_dict(data["deck_manager"])
    _deserialize_relics(data.get("relics", []))
    return true

func has_save() -> bool:
    return FileAccess.file_exists(SAVE_PATH)

func clear_save() -> void:
    if FileAccess.file_exists(SAVE_PATH):
        DirAccess.remove_absolute(
            ProjectSettings.globalize_path(SAVE_PATH))

func _serialize_relics() -> Array:
    var result := []
    for relic in GameManager.relics:
        result.append({"relic_name": relic.relic_name})
    return result

func _deserialize_relics(data: Array) -> void:
    GameManager.relics.clear()
    var pool := GameManager._build_relic_pool()
    for entry in data:
        for r in pool:
            if r.relic_name == entry["relic_name"]:
                GameManager.relics.append(r)
                break
```

func from_dict(data: Dictionary) -> void:
    clear()
    base_draw_count = data.get("base_draw_count", 5)
    var CardRes = load("res://resources/card_resource.gd")
    var EffRes = load("res://resources/effect_resource.gd")
    for cd in data.get("full_deck", []):
        var card := CardRes.new()
        card.card_name = cd["card_name"]
        card.owner_id = cd["owner_id"]
        card.cost = cd["cost"]
        card.play_animation = cd.get("play_animation", "idle")
        var effects := []
        for ed in cd.get("effects", []):
            var eff := EffRes.new()
            eff.effect_type = ed["effect_type"]
            eff.value = ed["value"]
            eff.target = ed.get("target", "SINGLE")
            effects.append(eff)
        card.effects = effects
        draw_pile.append(card)
    draw_pile.shuffle()

## Task 5: MapScene — 저장 감지 + 이어하기 UI

- [ ] **Step 1: MapScene._ready()에서 저장 파일 감지**

```gdscript
func _ready() -> void:
    if SaveManager.has_save() and GameManager.run_map.is_empty():
        _show_continue_dialog()
    elif GameManager.run_map.is_empty():
        GameManager.start_run()
    _build_map()

func _show_continue_dialog() -> void:
    # 간단한 확인 다이얼로그
    var dialog := ConfirmationDialog.new()
    dialog.dialog_text = "이전 런을 이어하시겠습니까?"
    dialog.confirmed.connect(_on_continue_confirmed)
    dialog.canceled.connect(_on_new_run)
    add_child(dialog)
    dialog.popup_centered()

func _on_continue_confirmed() -> void:
    SaveManager.load_save()
    _build_map()

func _on_new_run() -> void:
    SaveManager.clear_save()
    GameManager.start_run()
    _build_map()
```
